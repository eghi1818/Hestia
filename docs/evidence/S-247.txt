--- /etc/nginx/sites-available/hestia-ip.conf.asli ---
# Hestia Dashboard lewat port 80, hanya untuk permintaan dengan Host = IP VPS.
# Tidak mengganggu vhost waspadaweb.com karena server_name berbeda (exact match menang).
server {
    listen 80;
    listen [::]:80;
    server_name 93.127.133.160;

    client_max_body_size 64m;

    location / {
        proxy_pass http://127.0.0.1:8787;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_read_timeout 300s;
    }
}
